  // ── Open edit form ───────────────────────────────────────────────
  const handleEdit = (userId) => {
    const existing = payrolls.find(p => p.userId === userId);
    setEditingUserId(userId);
    setForm(existing ? {
      basicSalary:        String(existing.basicSalary        ?? ""),
      houseRentAllowance: String(existing.houseRentAllowance ?? ""),
      travelAllowance:    String(existing.travelAllowance    ?? ""),
      medicalAllowance:   String(existing.medicalAllowance   ?? ""),
      otherAllowances:    String(existing.otherAllowances    ?? ""),
      providentFund:      String(existing.providentFund      ?? ""),
      taxDeduction:       String(existing.taxDeduction       ?? ""),
      otherDeductions:    String(existing.otherDeductions    ?? ""),
      status:             existing.status,
    } : EMPTY_FORM);
    setTimeout(() => formRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }), 100);
  };
